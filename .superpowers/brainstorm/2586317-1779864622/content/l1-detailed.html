        """论文PDF/Markdown → 实体提取 → ontology对齐 → CC atoms+relations"""

        """自由文本 → BGE-M3粗筛 → LLM细提取 → ontology对齐 → CC"""

    def query(self, spec: QuerySpec) -> Subgraph:
        """统一查询入口。QuerySpec {atom_id?, keywords?, neighbor_depth?, filters?}"""

    def get_atom(self, atom_id: int) -> Atom | None: ...
    def get_relations(self, source_id: int | None = None, target_id: int | None = None,
                      rel_type: str | None = None, limit: int = 0) -> list[Relation]: ...

    def validate(self) -> ValidationReport:
